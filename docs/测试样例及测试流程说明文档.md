# 测试样例及测试流程说明文档

## 文档概述

本文档详细描述网络安全告警智能研判系统的测试样例设计、测试流程执行和测试结果分析。系统通过全面的测试验证确保各个模块和整体功能的正确性、稳定性和性能指标。

## 测试体系架构

### 测试框架组成

```
测试框架
├── 单元测试 (Unit Tests)
│   ├── 专家模型测试

├── 集成测试 (Integration Tests)

def test_response_time():
    """测试响应时间"""
    test_alert = {
        "attack_type": "SQL注入",
        "payload": "1' OR '1'='1",
        "attack_stage": "渗透尝试",
        "threat_level": "高风险"
    }

    response_times = []
    for i in range(50):
        start_time = time.time()
        response = requests.post("http://localhost:8001/analyze", json=test_alert)
        end_time = time.time()
        response_times.append(end_time - start_time)

    avg_response_time = sum(response_times) / len(response_times)
    max_response_time = max(response_times)

    print(f"平均响应时间: {avg_response_time:.2f}秒")
    print(f"最大响应时间: {max_response_time:.2f}秒")

    # 性能要求：平均响应时间 < 30秒
    assert avg_response_time < 30

    return True
```

**2.2 并发处理测试**
```python
def test_concurrent_processing():
    """测试并发处理能力"""
    import concurrent.futures

    def send_request():
        test_alert = {
            "attack_type": "SQL注入",
            "payload": "1' OR '1'='1",
            "attack_stage": "渗透尝试",
            "threat_level": "高风险"
        }
        response = requests.post("http://localhost:8001/analyze", json=test_alert)
        return response.json()

    # 并发发送10个请求
    with concurrent.futures.ThreadPoolExecutor(max_workers=10) as executor:
        futures = [executor.submit(send_request) for _ in range(10)]
        results = [future.result() for future in futures]

    # 验证所有请求都成功
    success_count = sum(1 for result in results if result.get("success", False))
    assert success_count == 10

    return True
```
